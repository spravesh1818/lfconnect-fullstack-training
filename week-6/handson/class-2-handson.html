src/models/todo.model.ts — add a document interface:

import { Schema, model } from "mongoose";

export interface Todo {
  title: string;
  completed: boolean;
}

const todoSchema = new Schema<Todo>(
  {
    title: { type: String, required: true },
    completed: { type: Boolean, default: false },
  },
  { timestamps: true }
);

export const TodoModel = model("Todo", todoSchema);

src/server.ts — type handlers and body:

import "dotenv/config";
import express, { Request, Response } from "express";

interface CreateTodoBody {
  title?: string;

const app = express();
app.use(express.json());

app.get("/", (_req: Request, res: Response) =>

app.get("/todos", async (_req: Request, res: Response) => {
  try {
    const todos = await TodoModel.find().sort({ createdAt: -1 });
    res.json(todos);
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: "Failed to fetch todos" });

app.post("/todos", async (req: Request, res: Response) => {
  try {
    const { title } = (req.body ?? {}) as CreateTodoBody;
    if (!title || typeof title !== "string" || !title.trim()) {
      return res.status(400).json({ error: "title is required" });

    const todo = await TodoModel.create({ title: title.trim() });
    res.status(201).json(todo);
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: "Failed to create todo" });
  }
});

Exercise 1 — Troubleshooting

  • An import path can only end with a '.ts' extension when 'allowImportingTsExtensions' is enabled — add "noEmit": true and "allowImportingTsExtensions": true to tsconfig.json.
  • Cannot find module './config/db.ts' — wrong path, or file not renamed from db.js.
  • Still importing .js — after rename, change import strings to .ts to match the file on disk.
  • req.body is any — cast with as CreateTodoBody until Zod middleware in Exercise 2.

Exercise 2 - Zod validation middleware

  1. Install Zod: npm install zod.
  2. Create src/middleware/validate.ts (reuse the Week 5 pattern).
  3. Create src/todo/schema/todo.schema.ts (we wire routes in Exercise 5).
// src/todo/schema/todo.schema.ts
import { z } from "zod";

export const createTodoSchema = z.object({
  title: z.string().min(1),
  completed: z.boolean().optional().default(false),
});

export const updateTodoSchema = createTodoSchema.partial();

export type CreateTodoDto = z.infer<typeof createTodoSchema>;
export type UpdateTodoDto = z.infer<typeof updateTodoSchema>;
// src/middleware/validate.ts
import { ZodSchema } from "zod";
import { Request, Response, NextFunction } from "express";

export const validate =
  (schema: ZodSchema) =>
  (req: Request, res: Response, next: NextFunction) => {
    const result = schema.safeParse(req.body);
    if (!result.success) {
      return res.status(400).json({ error: result.error.flatten() });
    }
    req.body = result.data;
    next();
  };
  1. Replace manual if (!title) on POST with validate(createTodoSchema).
  2. Send invalid JSON (empty title) and confirm 400.
Done when: valid POST creates a todo; invalid POST returns Zod error JSON with status 400.

Exercise 4 - PUT, PATCH, and DELETE

PUT = replace the whole resource. PATCH = update only fields you send — perfect for marking a todo complete.

// PUT — all fields required (use createTodoSchema)
app.put("/todos/:id", validate(createTodoSchema), async (req, res) => {
  const todo = await TodoModel.findByIdAndUpdate(
    req.params.id,
    req.body,
    { new: true, runValidators: true }
  );
  if (!todo) return res.status(404).json({ error: "Todo not found" });
  res.json(todo);
});

// PATCH — partial update (use updateTodoSchema)
app.patch("/todos/:id", validate(updateTodoSchema), async (req, res) => {
  const todo = await TodoModel.findByIdAndUpdate(
    req.params.id,
    req.body,
    { new: true, runValidators: true }
  );
  if (!todo) return res.status(404).json({ error: "Todo not found" });
  res.json(todo);
});

// DELETE
app.delete("/todos/:id", async (req, res) => {
  const deleted = await TodoModel.findByIdAndDelete(req.params.id);
  if (!deleted) return res.status(404).json({ error: "Todo not found" });
  res.status(204).send();
});
curl -X PUT http://localhost:3000/todos/ID \
  -H "Content-Type: application/json" \
  -d '{"title":"Finish Week 6 lab","completed":false}'

curl -X PATCH http://localhost:3000/todos/ID \
  -H "Content-Type: application/json" \
  -d '{"completed": true}'

curl -i -X DELETE http://localhost:3000/todos/ID
Done when: PUT replaces title and completed; PATCH with {"completed": true} marks done without changing the title; DELETE returns 204.

Exercise 5 - Refactor to todo/ domain folders

Organize by business area (todo domain), not one giant server.ts.

  server.ts

    model/todo.model.ts

Layer jobs (one sentence each):

  • routes — map URLs to controller functions
  • controller — read req, call service, send res
  • service — business logic + database calls
  • model — Mongoose schema
  • schema — Zod validation + DTO types
// src/todo/service/todo.service.ts
import { TodoModel } from "../model/todo.model.ts";
import type { CreateTodoDto, UpdateTodoDto } from "../schema/todo.schema.ts";

export async function listTodos() {
  return TodoModel.find().sort({ createdAt: -1 });
}

export async function createTodo(dto: CreateTodoDto) {
  return TodoModel.create(dto);
}

export async function getTodoById(id: string) {
  return TodoModel.findById(id);
}

export async function replaceTodo(id: string, dto: CreateTodoDto) {
  return TodoModel.findByIdAndUpdate(id, dto, { new: true, runValidators: true });
}

export async function updateTodo(id: string, dto: UpdateTodoDto) {
  return TodoModel.findByIdAndUpdate(id, dto, { new: true, runValidators: true });
}

export async function deleteTodo(id: string) {
  return TodoModel.findByIdAndDelete(id);
}
// src/todo/routes/todo.routes.ts
import { Router } from "express";
import { validate } from "../../middleware/validate.ts";
import { createTodoSchema, updateTodoSchema } from "../schema/todo.schema.ts";
import * as controller from "../controller/todo.controller.ts";

export const todoRoutes = Router();

todoRoutes.get("/", controller.list);
todoRoutes.post("/", validate(createTodoSchema), controller.create);
todoRoutes.get("/:id", controller.getOne);
todoRoutes.put("/:id", validate(createTodoSchema), controller.replace);
todoRoutes.patch("/:id", validate(updateTodoSchema), controller.update);
todoRoutes.delete("/:id", controller.remove);

Troubleshooting

  • Cannot find module after rename to .ts — use .ts in import paths and enable allowImportingTsExtensions in tsconfig.json.
  • CastError: Cast to ObjectId failed — invalid id format; return 400 with a friendly message.
  • PUT clears fields unexpectedly — PUT must send all required fields; use PATCH for partial updates like completed.
  • Routes 404 after refactor — check app.use("/todos", todoRoutes) and relative imports use .ts extensions.
  • Zod rejects empty title — use min(1); whitespace-only titles should fail validation.
  • Forgot express.json() — must run in app.ts before mounted routes.
