Exercise 1 - Extract router and app.ts

  1. Create src/app.ts — move express(), express.json(), health routes here.
  2. Create src/todo/routes/todo.routes.ts with Router().
  3. Move GET/POST /todos from server.ts into the router (handlers can stay inline for now).
  4. Mount: app.use("/todos", todoRoutes).
  5. Slim down server.ts to dotenv + connectDB() + app.listen.
// src/todo/routes/todo.routes.ts
import { Router } from "express";
import { validate } from "../../middleware/validate.ts";
import { createTodoSchema } from "../schema/todo.schema.ts";
import { TodoModel } from "../../models/todo.model.ts";

export const todoRoutes = Router();

todoRoutes.get("/", async (_req, res) => {
  const todos = await TodoModel.find().sort({ createdAt: -1 });
  res.json(todos);
});

todoRoutes.post("/", validate(createTodoSchema), async (req, res) => {
  const todo = await TodoModel.create(req.body);
  res.status(201).json(todo);
});
// src/app.ts
import express from "express";
import { todoRoutes } from "./todo/routes/todo.routes.ts";

export const app = express();
app.use(express.json());

app.get("/", (_req, res) => res.json({ message: "hello from todo-api" }));
app.get("/health", (_req, res) => res.json({ status: "ok" }));
app.use("/todos", todoRoutes);
Done when: GET/POST /todos still work via the mounted router.

Exercise 3 - Full CRUD routes

Add remaining routes in todo.routes.ts with validation where needed.

todoRoutes.get("/:id", controller.getOne);
todoRoutes.put("/:id", validate(createTodoSchema), controller.replace);
todoRoutes.patch("/:id", validate(updateTodoSchema), controller.update);
todoRoutes.delete("/:id", controller.remove);
// Service additions
export async function getTodoById(id: string) {
  return TodoModel.findById(id);
}

export async function replaceTodo(id: string, dto: CreateTodoDto) {
  return TodoModel.findByIdAndUpdate(id, dto, { new: true, runValidators: true });
}

export async function updateTodo(id: string, dto: UpdateTodoDto) {
  return TodoModel.findByIdAndUpdate(id, dto, { new: true, runValidators: true });
}

export async function deleteTodo(id: string) {
  return TodoModel.findByIdAndDelete(id);
}

Controllers should return 404 when service returns null, 204 on delete, 400 on invalid id.

Done when: all five CRUD operations work; PATCH can set completed: true without changing title.

Troubleshooting

  • Routes 404 after refactor — check app.use("/todos", todoRoutes) and .ts import paths.
  • CastError: Cast to ObjectId failed — invalid id; return 400 with a friendly message.
  • PUT clears fields unexpectedly — send all required fields on PUT; use PATCH for partial updates.
  • Forgot express.json() — must run in app.ts before mounted routes.
