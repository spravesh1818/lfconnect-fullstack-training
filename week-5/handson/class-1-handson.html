Exercise 1 - TypeScript + Express project setup

  1. Initialize the project: npm init -y. Edit package.json and add "type": "module".
  2. Install runtime deps: npm install express.
  3. Install dev deps: npm install -D typescript tsx @types/node @types/express.
  4. Add a tsconfig.json:
{
  "compilerOptions": {
    "target": "ES2022",
    "module": "NodeNext",
    "moduleResolution": "NodeNext",
    "strict": true,
    "outDir": "dist",
    "esModuleInterop": true,
    "skipLibCheck": true
  },
  "include": ["src"]
}
  1. Create src/server.ts with a hello-world endpoint and a /health endpoint.
  2. Add a script to package.json: "dev": "tsx watch src/server.ts".
  3. Run npm run dev and verify both endpoints in the browser or with curl.
// src/server.ts
import express from "express";

const app = express();
app.use(express.json());

app.get("/", (_req, res) => res.json({ message: "hello world" }));
app.get("/health", (_req, res) => res.json({ status: "ok" }));

app.listen(3000, () => console.log("listening on http://localhost:3000"));
Done when: GET / returns the hello-world JSON and GET /health returns { status: "ok" }.

Exercise 2 - Meetings endpoints (in-memory)

  1. Add a Meeting type and an in-memory meetings array.
  2. Implement GET /meetings to return the array.
  3. Implement POST /meetings to create a new meeting from req.body.
  4. Use crypto.randomUUID() for the id.
  5. Test with curl: 
    curl -X POST http://localhost:3000/meetings \
  -H "Content-Type: application/json" \
  -d '{"title":"Sprint","date":"2026-06-01T10:00:00Z","host":"alice"}'
type Meeting = {
  id: string;
  title: string;
  date: string;
  host: string;
};

const meetings: Meeting[] = [];

app.get("/meetings", (_req, res) => res.json(meetings));

app.post("/meetings", (req, res) => {
  const meeting: Meeting = { id: crypto.randomUUID(), ...req.body };
  meetings.push(meeting);
  res.status(201).json(meeting);
});
Done when: POST adds a meeting and a subsequent GET returns it in the array.

Exercise 3 - Zod validation middleware

  1. Install Zod: npm install zod.
  2. Create src/schemas/meeting.ts with a createMeetingSchema.
  3. Create src/middleware/validate.ts with a reusable validate middleware that uses safeParse.
  4. Apply validate(createMeetingSchema) on POST /meetings.
  5. Send an invalid POST (e.g. missing title) and confirm a 400 response with a readable error.
// src/schemas/meeting.ts
import { z } from "zod";

export const createMeetingSchema = z.object({
  title: z.string().min(3),
  date: z.string().datetime(),
  host: z.string().min(1),
  participants: z.array(z.string().email()).default([]),
});
// src/middleware/validate.ts
import { ZodSchema } from "zod";
import { Request, Response, NextFunction } from "express";

export const validate =
  (schema: ZodSchema) =>
  (req: Request, res: Response, next: NextFunction) => {
    const result = schema.safeParse(req.body);
    if (!result.success) {
      return res.status(400).json({ error: result.error.flatten() });
    }
    req.body = result.data;
    next();
  };
Done when: valid payloads create meetings; invalid payloads return a 400 with a Zod error shape.

Exercise 4 - DTO + custom logger + error middleware

  1. Export a DTO type from the schema: export type CreateMeetingDto = z.infer<typeof createMeetingSchema>.
  2. Use CreateMeetingDto in the route handler instead of req.body: any.
  3. Add a custom request logger middleware that prints method url status durationMs.
  4. Add a final error-handling middleware with the 4-arg signature.
  5. Throw a sample error from a route (e.g. GET /boom) and confirm the error middleware returns a JSON 500.
// src/middleware/logger.ts
import { Request, Response, NextFunction } from "express";

export function requestLogger(req: Request, res: Response, next: NextFunction) {
  const start = Date.now();
  res.on("finish", () => {
    console.log(`${req.method} ${req.url} ${res.statusCode} ${Date.now() - start}ms`);
  });
  next();
}
// src/middleware/error.ts
import { Request, Response, NextFunction } from "express";

export function errorHandler(err: unknown, _req: Request, res: Response, _next: NextFunction) {
  console.error(err);
  res.status(500).json({ error: "Internal Server Error" });
}
// src/server.ts (order matters)
app.use(express.json());
app.use(requestLogger);

app.get("/boom", () => { throw new Error("kaboom"); });

app.post("/meetings", validate(createMeetingSchema), (req, res) => {
  const dto = req.body as CreateMeetingDto;
  const meeting: Meeting = { id: crypto.randomUUID(), ...dto };
  meetings.push(meeting);
  res.status(201).json(meeting);
});

app.use(errorHandler); // last
Done when: the logger prints one line per request, GET /boom returns a 500 JSON from the error handler, and the handler uses the DTO type.

Lab checklist

  • Server boots with npm run dev and logs the listen URL.
  • GET / and GET /health respond with JSON.
  • GET /meetings and POST /meetings work end-to-end.
  • Invalid POST /meetings returns a 400 with a Zod error.
  • Request logger prints a line for every request.
  • Error middleware turns thrown errors into JSON 500s.
  • DTO type is reused in the handler (no any).
